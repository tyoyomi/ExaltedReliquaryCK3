# Exalted Reliquary — artifact templates (additive; do not overwrite vanilla)

er_reliquary_template = {
	can_equip = {
		always = yes
	}

	can_benefit = {
		always = yes
	}

	fallback = {
		court_grandeur_baseline_add = 1
	}
}

er_regalia_template = {
	can_equip = {
		trigger_if = {
			limit = {
				has_ach_dlc_trigger = yes
				coronation_proper_artifact_crown_trigger = yes
				NOT = { scope:artifact = { category = court } }
			}
			NOT = { has_realm_law = uncrowned }
		}
	}

	can_benefit = {
		always = yes
	}

	fallback = {
		court_grandeur_baseline_add = 2
		monthly_prestige = 0.1
	}
}

er_heirloom_template = {
	can_equip = {
		always = yes
	}

	can_benefit = {
		OR = {
			NOT = { scope:artifact = { exists = var:er_heirloom_house } }
			AND = {
				exists = house
				house = scope:artifact.var:er_heirloom_house
			}
		}
	}

	fallback = {
		court_grandeur_baseline_add = 1
		monthly_prestige = 0.05
	}
}
